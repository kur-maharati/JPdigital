/* ========================================================= PENGATURAN ========================================================= */ const AUTO_REFRESH = 30000; let waktuSelesai = "-"; let sedangMemuat = false; let refreshTransisiTerakhir = 0; /* ========================================================= UPDATE JAM DIGITAL ========================================================= */ function updateClock() { const sekarang = new Date(); } finally { sedangMemuat = false; } } /* ========================================================= TAMPILKAN DATA ========================================================= */ function tampilkanJadwal(data) { /* ----------------------------------------------------- NAMA SEKOLAH ----------------------------------------------------- */ if ( data.setting && data.setting.NamaSekolah ) { const namaSekolah = document.getElementById( "namaSekolah" ); if (namaSekolah) { namaSekolah.textContent = data.setting.NamaSekolah; } } /* ----------------------------------------------------- JAM AKTIF ----------------------------------------------------- */ if (data.jam) { const jamKe = document.getElementById( "jamKe" ); const namaJam = document.getElementById( "namaJam" ); const jamSelesai = document.getElementById( "jamSelesai" ); if (jamKe) { jamKe.textContent = data.jam.jamKe || "-"; } if (namaJam) { namaJam.textContent = data.jam.namaJam || "-"; } if (jamSelesai) { jamSelesai.textContent = data.jam.selesai || "-"; } waktuSelesai = data.jam.selesai || "-"; } /* ----------------------------------------------------- TABEL JADWAL ----------------------------------------------------- */ const tbody = document.getElementById( "dataJadwal" ); if (!tbody) { return; } let html = ""; if ( Array.isArray(data.jadwal) ) { data.jadwal.forEach( function(item) { const sekarang = item.sekarang || {}; const berikutnya = item.berikutnya || {}; html += ` ${escapeHtml( item.kelas || "-" )} ${escapeHtml( sekarang.mapel || "-" )} ${escapeHtml( sekarang.guru || "-" )} ${escapeHtml( berikutnya.mapel || "-" )} ${escapeHtml( berikutnya.guru || "-" )} `; } ); } if (html === "") { html = ` Tidak ada data jadwal `; } tbody.innerHTML = html; } /* ========================================================= ESCAPE HTML ========================================================= */ function escapeHtml(text) { return String(text) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } /* ========================================================= KONVERSI HH:mm KE MENIT ========================================================= */ function waktuKeMenit(waktu) { if ( !waktu || waktu === "-" ) { return null; } const bagian = String(waktu).split(":"); if (bagian.length < 2) { return null; } const jam = Number(bagian[0]); const menit = Number(bagian[1]); if ( isNaN(jam) || isNaN(menit) ) { return null; } return ( jam * 60 + menit ); } /* ========================================================= COUNTDOWN ========================================================= */ function updateCountdown() { const countdown = document.getElementById( "countdown" ); if (!countdown) { return; } if ( !waktuSelesai || waktuSelesai === "-" ) { countdown.textContent = "00:00:00"; return; } const targetMenit = waktuKeMenit( waktuSelesai ); if (targetMenit === null) { countdown.textContent = "00:00:00"; return; } const sekarang = new Date(); const sekarangMenit = sekarang.getHours() * 60 + sekarang.getMinutes(); const sekarangDetik = sekarang.getSeconds(); let selisihDetik = ( targetMenit - sekarangMenit ) * 60 - sekarangDetik; if (selisihDetik < 0) { selisihDetik = 0; } const jam = Math.floor( selisihDetik / 3600 ); const menit = Math.floor( ( selisihDetik % 3600 ) / 60 ); const detik = selisihDetik % 60; countdown.textContent = String(jam) .padStart(2, "0") + ":" + String(menit) .padStart(2, "0") + ":" + String(detik) .padStart(2, "0"); /* ----------------------------------------------------- REFRESH SAAT JAM BERAKHIR ----------------------------------------------------- */ if (selisihDetik === 0) { const sekarangTimestamp = Date.now();
